\section{Convergence Theory for QR Box-Dual FedPDHG}
\label{sec:convergence-theory}

We consider the penalized quantile-regression problem
\begin{equation}
    \min_{\beta \in \mathbb{R}^p}
    F(\beta)
    :=
    \frac{1}{n}\sum_{i=1}^n
    \rho_\tau(y_i - x_i^\top \beta)
    + r(\beta),
    \label{eq:qr-primal}
\end{equation}
where $\rho_\tau(u)=u(\tau-\mathbf{1}\{u<0\})$ is the check loss and
$r$ is a proper closed convex penalty. The key structural fact is the
Fenchel box-dual representation
\begin{equation}
    \rho_\tau(u)
    =
    \max_{v \in [\tau-1,\tau]} vu .
\end{equation}
Thus \eqref{eq:qr-primal} can be written as the convex-concave saddle problem
\begin{equation}
    \min_{\beta \in \mathbb{R}^p}
    \max_{v \in \mathcal{V}}
    \mathcal{L}(\beta,v)
    :=
    r(\beta)
    +
    \frac{1}{n}v^\top (y-X\beta),
    \qquad
    \mathcal{V}:=[\tau-1,\tau]^n .
    \label{eq:qr-saddle}
\end{equation}
The proposed QR box-dual method is a primal-dual hybrid gradient method applied
to \eqref{eq:qr-saddle}, with dual projection onto the box $\mathcal{V}$ and a
proximal step for $r$.

\subsection{Deterministic Full-Participation Rate}

We first state the deterministic benchmark. This corresponds to full client
participation and full local dual refresh at every communication round.

\paragraph{Assumption D.}
The penalty $r$ is proper, closed, and convex. The covariates are bounded, and
the matrix $X$ has finite spectral norm. The step sizes $\eta,\sigma>0$ satisfy
\begin{equation}
    \eta\sigma \frac{\|X\|_2^2}{n^2} < 1 .
    \label{eq:pdhg-stepsize}
\end{equation}

\begin{theorem}[Deterministic ergodic primal-dual gap]
\label{thm:deterministic-pdhg}
Let $(\beta^t,v^t)$ be generated by the deterministic QR box-dual PDHG method
with step sizes satisfying \eqref{eq:pdhg-stepsize}. Define the ergodic
averages
\begin{equation}
    \bar{\beta}^T=\frac{1}{T}\sum_{t=1}^T \beta^t,
    \qquad
    \bar{v}^T=\frac{1}{T}\sum_{t=1}^T v^t .
\end{equation}
For any saddle point $(\beta^\star,v^\star)$ of \eqref{eq:qr-saddle},
\begin{equation}
    \mathcal{G}(\bar{\beta}^T,\bar{v}^T)
    :=
    \mathcal{L}(\bar{\beta}^T,v^\star)
    -
    \mathcal{L}(\beta^\star,\bar{v}^T)
    \le
    \frac{1}{T}
    \left(
        \frac{\|\beta^\star-\beta^0\|^2}{2\eta}
        +
        \frac{\|v^\star-v^0\|^2}{2\sigma}
    \right).
    \label{eq:deterministic-rate}
\end{equation}
Consequently, the deterministic full-participation method has ergodic
primal-dual gap rate $\mathcal{O}(1/T)$.
\end{theorem}

\begin{proof}
The saddle objective \eqref{eq:qr-saddle} has the standard form
\[
    r(\beta) + \langle K\beta,v\rangle - h^\ast(v),
    \qquad
    K=-X/n,\quad
    h^\ast(v)= -v^\top y/n + \iota_{\mathcal{V}}(v),
\]
where $\iota_{\mathcal{V}}$ is the indicator of the box $\mathcal{V}$. The
dual update is the proximal step of $h^\ast$, which is exactly projection onto
$\mathcal{V}$ after the affine shift by $y/n$, and the primal update is the
proximal step of $r$. Under
$\eta\sigma\|K\|_2^2<1$, the standard PDHG one-step inequality gives, for all
$(\beta,v)$,
\begin{align}
    \mathcal{L}(\beta^{t+1},v)-\mathcal{L}(\beta,v^{t+1})
    &\le
    \frac{1}{2\eta}
    \left(
        \|\beta-\beta^t\|^2-\|\beta-\beta^{t+1}\|^2
    \right) \nonumber\\
    &\quad+
    \frac{1}{2\sigma}
    \left(
        \|v-v^t\|^2-\|v-v^{t+1}\|^2
    \right).
    \label{eq:pdhg-one-step}
\end{align}
Summing \eqref{eq:pdhg-one-step} from $t=0$ to $T-1$ telescopes the squared
distance terms. Setting $(\beta,v)=(\beta^\star,v^\star)$ and using convexity
in $\beta$ and concavity in $v$ to pass from the average of gaps to the gap at
the ergodic averages yields \eqref{eq:deterministic-rate}.
\end{proof}

\subsection{Stochastic Federated Rate Without Variance Reduction}

We now consider the actual federated setting. At each round, a subset of
clients is sampled, and each selected client updates only a mini-batch of its
sample-level dual variables. Unselected clients contribute cached directions.
Let $z^t=(\beta^t,v^t)$ and let $\widehat{g}^t$ denote the stochastic cached
primal direction used by the server.

\paragraph{Assumption S.}
There exist constants $R,V,\Sigma,\Delta<\infty$ such that:
\begin{enumerate}
    \item $\|x_i\|\le R$ for all $i$;
    \item the dual variables remain in the box
    $v_i^t\in[\tau-1,\tau]$, hence $|v_i^t|\le V:=\max\{\tau,1-\tau\}$;
    \item conditional on the history $\mathcal{F}_t$, the stochastic direction
    has bounded second moment,
    \begin{equation}
        \mathbb{E}\big[
        \|\widehat{g}^t-\mathbb{E}(\widehat{g}^t\mid\mathcal{F}_t)\|^2
        \mid\mathcal{F}_t
        \big]\le \Sigma^2;
        \label{eq:bounded-variance}
    \end{equation}
    \item the cache bias caused by partial participation and stale local
    directions is bounded by
    \begin{equation}
        \big\|
        \mathbb{E}(\widehat{g}^t\mid\mathcal{F}_t)-g^t
        \big\|
        \le \Delta_t,
        \qquad
        \frac{1}{T}\sum_{t=0}^{T-1}\Delta_t
        \le \bar{\Delta}_T ,
        \label{eq:stale-bias}
    \end{equation}
    where $g^t$ is the exact full-participation primal direction;
    \item the iterates stay in a bounded level set with diameter at most $D$.
\end{enumerate}

\begin{theorem}[Stochastic federated ergodic rate without VR]
\label{thm:stochastic-rate}
Suppose Assumption S holds. Let the stochastic federated QR box-dual method use
a constant step size $\eta_t=\sigma_t=\gamma/\sqrt{T}$ for $T$ rounds, with
$\gamma$ small enough to satisfy the PDHG stability condition. Then the
ergodic primal-dual gap satisfies
\begin{equation}
    \mathbb{E}\big[
    \mathcal{G}(\bar{\beta}^T,\bar{v}^T)
    \big]
    \le
    \frac{C_1}{\sqrt{T}}
    +
    C_2 \bar{\Delta}_T,
    \label{eq:stochastic-rate}
\end{equation}
where $C_1$ depends on $D$, $\gamma$, and $\Sigma^2$, and $C_2$ depends on the
bounded level-set diameter. If the average cache bias satisfies
$\bar{\Delta}_T=\mathcal{O}(T^{-1/2})$, then
\begin{equation}
    \mathbb{E}\big[
    \mathcal{G}(\bar{\beta}^T,\bar{v}^T)
    \big]
    =
    \mathcal{O}(T^{-1/2}).
\end{equation}
\end{theorem}

\begin{proof}
The proof follows the deterministic PDHG descent inequality with an additional
stochastic error term. Replacing the exact primal direction $g^t$ by
$\widehat{g}^t$ adds the inner product
\[
    \left\langle
    \widehat{g}^t-g^t,\,
    \beta^{t+1}-\beta
    \right\rangle .
\]
Decompose this error as
\[
    \widehat{g}^t-g^t
    =
    \underbrace{
    \widehat{g}^t-\mathbb{E}(\widehat{g}^t\mid\mathcal{F}_t)
    }_{\text{martingale noise}}
    +
    \underbrace{
    \mathbb{E}(\widehat{g}^t\mid\mathcal{F}_t)-g^t
    }_{\text{cache bias}} .
\]
The martingale-noise term has zero conditional mean and bounded second moment
by \eqref{eq:bounded-variance}. Applying Young's inequality gives a cumulative
contribution bounded by
\[
    \sum_{t=0}^{T-1} \eta_t \Sigma^2 .
\]
The cache-bias term is controlled by the bounded diameter assumption and
\eqref{eq:stale-bias}, yielding a cumulative contribution of order
\[
    D\sum_{t=0}^{T-1}\Delta_t .
\]
After summing the one-step inequalities and dividing by
$\sum_{t=0}^{T-1}\eta_t$, the deterministic telescoping part contributes
$\mathcal{O}(1/(\eta T))$, the stochastic variance contributes
$\mathcal{O}(\eta\Sigma^2)$, and the stale-cache bias contributes
$\mathcal{O}(\bar{\Delta}_T)$. With $\eta=\gamma/\sqrt{T}$ this gives
\eqref{eq:stochastic-rate}.
\end{proof}

\paragraph{Fixed-step interpretation.}
If the method is run with a fixed small step size $\eta=\sigma$, the same
argument gives the neighborhood bound
\begin{equation}
    \mathbb{E}\big[
    \mathcal{G}(\bar{\beta}^T,\bar{v}^T)
    \big]
    \le
    \frac{C_3}{\eta T}
    +
    C_4 \eta \Sigma^2
    +
    C_5 \bar{\Delta}_T .
    \label{eq:fixed-step-rate}
\end{equation}
Thus a fixed-step stochastic method converges at an $\mathcal{O}(1/T)$
transient rate to a variance- and staleness-dependent neighborhood.

\subsection{Effect of Staleness and Client-Robust Weighting}

The staleness-aware and robust variants use client weights $w_j^t$ in the
server aggregation. The above rate remains valid if the weights are uniformly
bounded and normalized:
\begin{equation}
    0 < w_{\min} \le w_j^t \le w_{\max}<1,
    \qquad
    \sum_{j=1}^m w_j^t = 1 .
\end{equation}
For sample weighting, the limiting objective is the usual empirical risk.
For uniform or custom client weighting, the same proof applies to the
corresponding client-weighted quantile objective
\begin{equation}
    F_w(\beta)
    =
    \sum_{j=1}^m w_j
    \frac{1}{n_j}\sum_{i\in\mathcal{D}_j}
    \rho_\tau(y_i-x_i^\top\beta)
    + r(\beta).
\end{equation}
For adaptive weighting, one can view the method as solving a slowly varying
sequence of client-weighted saddle problems. If the adaptive weights are
bounded and have finite total variation,
\begin{equation}
    \sum_{t=0}^{T-1}\|w^{t+1}-w^t\|_1 < \infty ,
\end{equation}
then the additional error enters the bound as a lower-order variation term.

\subsection{Variance Reduction Remark}

The control-variate variant is not required for convergence. Its role is to
reduce the variance constant $\Sigma^2$ in \eqref{eq:bounded-variance}. If the
corrected direction $\widehat{g}_{\mathrm{vr}}^t$ satisfies
\begin{equation}
    \mathbb{E}
    \big[
    \|\widehat{g}_{\mathrm{vr}}^t
    -
    \mathbb{E}(\widehat{g}_{\mathrm{vr}}^t\mid\mathcal{F}_t)\|^2
    \mid \mathcal{F}_t
    \big]
    \le
    \Sigma_{\mathrm{vr}}^2
    \le
    \Sigma^2 ,
\end{equation}
then Theorem~\ref{thm:stochastic-rate} holds with $\Sigma_{\mathrm{vr}}^2$ in
place of $\Sigma^2$. Therefore variance reduction improves the constant in the
$\mathcal{O}(T^{-1/2})$ stochastic rate, but the rate order remains
$\mathcal{O}(T^{-1/2})$ under the general nonsmooth federated setting.
